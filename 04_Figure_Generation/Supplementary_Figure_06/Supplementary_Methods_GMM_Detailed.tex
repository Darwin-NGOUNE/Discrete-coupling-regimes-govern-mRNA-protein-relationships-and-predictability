\subsection*{1. Objective and Input Data}
To determine whether the mRNA--protein response regimes identified by Differential Pattern Analysis (DiPa) reflect distinct biological subpopulations rather than arbitrary slices of a continuous cloud, we fitted constrained bivariate Gaussian Mixture Models (GMM) to the empirical log$_2$ fold changes independently of the $\pm 0.5$ cutoffs. 

The input is a matrix $X \in \mathbb{R}^{N \times 2}$ ($N = 943$ core matched pairs), where for each gene $i$:
\begin{equation}
X_i = \begin{pmatrix} x_i \\ y_i \end{pmatrix} = \begin{pmatrix} \log_2\text{FC}_{\text{RNA}, i} \\ \log_2\text{FC}_{\text{Prot}, i} \end{pmatrix}.
\end{equation}

\subsection*{2. Model Specification and Biophysical Constraints}
The probability density of observation $X_i$ under a mixture of $K \in \{1, 2, 3\}$ bivariate Gaussians is:
\begin{equation}
p(X_i \mid \theta) = \sum_{k=1}^K \pi_k \, \mathcal{N}_2\left(X_i \mid \mu_k, \, \Sigma_k\right), \quad \text{with } \pi_k \ge 0, \quad \sum_{k=1}^K \pi_k = 1,
\end{equation}
where $\mu_k = (\mu_{xk}, \mu_{yk})^\top$ are component centroids and $\Sigma_k = \begin{pmatrix} \sigma_{xk}^2 & \rho_k \sigma_{xk}\sigma_{yk} \\ \rho_k \sigma_{xk}\sigma_{yk} & \sigma_{yk}^2 \end{pmatrix}$ are covariance matrices.

To prevent unconstrained components from fitting arbitrary noise, three biophysical constraints were enforced:
\begin{enumerate}[noitemsep,topsep=2pt]
    \item \textbf{Component 1 (Diagonal / Co-regulation):} Represents canonical stoichiometric coupling. The correlation is bounded within a biologically realistic coupling window:
    \begin{equation}
    \rho_1 \in [0.70, \, 0.90].
    \end{equation}
    \item \textbf{Component 2 (Vertical / Protein-dominant decoupling):} Represents protein changes occurring without significant transcriptomic shifts. Coordinate correlation is set to zero, and variance is vertically elongated by factor $c$:
    \begin{equation}
    \rho_2 = 0, \qquad \sigma_{y2}^2 \ge c \cdot \sigma_{x2}^2.
    \end{equation}
    \item \textbf{Component 3 (Horizontal / Post-transcriptional buffering, for $K=3$):} Represents large mRNA changes buffered at the protein level. Correlation is set to zero, and variance is horizontally elongated:
    \begin{equation}
    \rho_3 = 0, \qquad \sigma_{x3}^2 \ge c \cdot \sigma_{y3}^2.
    \end{equation}
\end{enumerate}
The elongation factor was systematically tested across $c \in \{1.3, 1.5, 2.0, 2.5, 4.0\}$ (aspect ratios $\sqrt{c} \in [1.14, 2.00]$).

\subsection*{3. Expectation-Maximization (EM) Algorithm}
Parameters were estimated iteratively via the Expectation-Maximization (EM) algorithm~\cite{dempster1977maximum, mclachlan2000finite} until the log-likelihood converged ($|\ell^{(t+1)} - \ell^{(t)}| < 10^{-6}$):
\begin{itemize}[noitemsep,topsep=2pt]
    \item \textbf{Initialization:} Posterior weights $\gamma_{ik}^{(0)} \in [0, 1]$ were initialized deterministically based on coarse regions (vertical candidates: $|x_i| < 0.55, |y_i| > 0.40$; horizontal candidates: $|x_i| > 0.50, |y_i| < 0.50$; diagonal otherwise), normalized so $\sum_k \gamma_{ik}^{(0)} = 1$.
    \item \textbf{M-Step (Maximization):} Given weights $\gamma_{ik}$, component sample sizes $N_k = \sum_i \gamma_{ik}$ and mixing weights $\pi_k = N_k / N$ were computed. Centroids and sample dispersions were updated:
    \begin{equation}
    \mu_k = \frac{1}{N_k}\sum_{i=1}^N \gamma_{ik} X_i, \quad v_{xk} = \frac{1}{N_k}\sum_{i=1}^N \gamma_{ik}(x_i - \mu_{xk})^2, \quad v_{yk} = \frac{1}{N_k}\sum_{i=1}^N \gamma_{ik}(y_i - \mu_{yk})^2,
    \end{equation}
    and $c_{xy,1} = \frac{1}{N_1}\sum_{i=1}^N \gamma_{i1}(x_i - \mu_{x1})(y_i - \mu_{y1})$. Constraints were then directly applied to define the final dispersion parameters $\hat{\sigma}^2$ and matrices $\Sigma_k$:
    \begin{itemize}[noitemsep,topsep=1pt]
        \item For $k=1$: $\hat{\sigma}_{x1}^2 = v_{x1}$, $\hat{\sigma}_{y1}^2 = v_{y1}$, $\hat{\rho}_1 = \min(0.90, \max(0.70, c_{xy,1}/\sqrt{v_{x1}v_{y1}}))$, and $\Sigma_1 = \begin{pmatrix} \hat{\sigma}_{x1}^2 & \hat{\rho}_1 \hat{\sigma}_{x1}\hat{\sigma}_{y1} \\ \hat{\rho}_1 \hat{\sigma}_{x1}\hat{\sigma}_{y1} & \hat{\sigma}_{y1}^2 \end{pmatrix}$.
        \item For $k=2$: $\hat{\sigma}_{x2}^2 = v_{x2}$, $\hat{\sigma}_{y2}^2 = \max(v_{y2}, \, c \cdot v_{x2})$, and $\Sigma_2 = \operatorname{diag}(\hat{\sigma}_{x2}^2, \, \hat{\sigma}_{y2}^2)$.
        \item For $k=3$: $\hat{\sigma}_{x3}^2 = \max(v_{x3}, \, c \cdot v_{y3})$, $\hat{\sigma}_{y3}^2 = v_{y3}$, and $\Sigma_3 = \operatorname{diag}(\hat{\sigma}_{x3}^2, \, \hat{\sigma}_{y3}^2)$.
    \end{itemize}
    \item \textbf{E-Step (Expectation):} Using the updated parameters, posterior responsibilities were recalculated:
    \begin{equation}
    \gamma_{ik}^{(t+1)} = \frac{\pi_k \, \mathcal{N}_2\left(X_i \mid \mu_k, \, \Sigma_k\right)}{\sum_{j=1}^K \pi_j \, \mathcal{N}_2\left(X_i \mid \mu_j, \, \Sigma_j\right)}, \quad \ell^{(t+1)} = \sum_{i=1}^N \ln \left( \sum_{k=1}^K \pi_k \, \mathcal{N}_2\left(X_i \mid \mu_k, \, \Sigma_k\right) \right).
    \end{equation}
\end{itemize}

\subsection*{4. Model Selection (BIC)}
Model selection across $K \in \{1, 2, 3\}$ and $c \in \{1.3, 1.5, 2.0, 2.5, 4.0\}$ was evaluated using the Bayesian Information Criterion (BIC)~\cite{schwarz1978estimating}:
\begin{equation}
\text{BIC} = -2\hat{\ell} + P \ln(N),
\end{equation}
where $\hat{\ell}$ is the maximized log-likelihood, $N = 943$ ($\ln(943) \approx 6.849$), and $P$ is the number of free parameters:
\begin{itemize}[noitemsep,topsep=2pt]
    \item $K = 1$: 2 means + 2 variances + 1 covariance $\implies P = 5$.
    \item $K = 2$ (constrained): 1 weight ($\pi_1$) + 4 means + 3 dispersion (Comp 1) + 2 variances (Comp 2) $\implies P = 10$.
    \item $K = 3$ (constrained): 2 weights + 6 means + 3 (Comp 1) + 2 (Comp 2) + 2 (Comp 3) $\implies P = 15$.
\end{itemize}
A difference $\Delta\text{BIC} < -10$ provides decisive evidence in favor of the lower-BIC model on the Kass--Raftery scale~\cite{kass1995bayes}. BIC was chosen over classical likelihood-ratio tests because testing component complexity ($K$ vs.\ $K+1$) places the null hypothesis ($\pi_{K+1} = 0$) on the boundary of the probability simplex, violating Wilks' regularity conditions~\cite{mclachlan2000finite}.

\subsection*{5. Confidence Ellipses}
Concentration ellipses for component $k$ at confidence levels $\alpha \in \{0.50, 0.90, 0.95\}$ were generated using the spectral decomposition $\hat{\Sigma}_k = V \Lambda V^\top$:
\begin{equation}
E_k(\alpha) = \hat{\mu}_k + \sqrt{F_{\chi_2^2}^{-1}(\alpha)} \cdot V \Lambda^{1/2} \begin{pmatrix} \cos \phi \\ \sin \phi \end{pmatrix}, \quad \phi \in [0, 2\pi),
\end{equation}
where $V$ is the orthogonal matrix of eigenvectors governing the spatial orientation (rotation) of the ellipse axes, $\Lambda = \operatorname{diag}(\lambda_1, \lambda_2)$ contains the eigenvalues (variances along principal axes), and $\sqrt{F_{\chi_2^2}^{-1}(\alpha)}$ is the Mahalanobis critical radius derived from the $\chi_2^2$ distribution enclosing probability mass $\alpha$ ($r_{0.50} \approx 1.177$, $r_{0.90} \approx 2.146$, and $r_{0.95} \approx 2.448$).
